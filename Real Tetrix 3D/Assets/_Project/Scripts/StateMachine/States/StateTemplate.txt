using UnityEngine;

namespace RafaEslava {

public class StateSettings: IState
{
    public void Tick() 
    {
        //Debug.Log("Tetrix State Settings Tick *****************");
    }

    public void OnEnter()
    {
        Debug.Log("Tetrix State Settings Enter *****************");   

        Tools.IsPaused = true;
        Events.OnSettingsEnter?.Invoke();                    
    }      
    
    public void OnExit()
    {
        Debug.Log("Tetrix State Settings Exit *****************");              		
        Tools.IsPaused = false;
    }      
}

}
